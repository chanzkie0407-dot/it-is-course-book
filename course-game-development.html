What Is Canvas?

Ano ang Canvas?

Think of it as a digital drawing board — you draw shapes, colors, and move everything with code, every moment. Parang pisara — gumuhit gamit ang code, galawin bawat sandali

Basic Template:

Unang halimbawa — kopyahin at subukan 
<!DOCTYPE html>
<html>
<body>
  <canvas id="gameCanvas" width="800" height="600"></canvas>
  <script>
    const canvas = document.getElementById('gameCanvas');
    const ctx = canvas.getContext('2d'); // drawing tool ✅
    
    // Draw a circle
    ctx.beginPath();
    ctx.arc(100, 100, 30, 0, Math.PI * 2); // x, y, radius, start, end
    ctx.fillStyle = '#10b981';
    ctx.fill();
    ctx.closePath();
  </script>
</body>
</html>

Save as game.html → Open in browser → You'll see a green circle! ✅ I-save bilang game.html → buksan sa browser → may luntian na bilog

📖 Chapter 2 — The Game Loop ❗

Ang Pangkalahatang Pag-ikot — pinakamahalagang bahagi

This is the heart of every game — runs continuously, hundreds of times per second!

Tumatakbo nang paulit-ulit — daan-daang beses bawat segundo 
function gameLoop() {
  // 1. Clear screen
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  // 2. Update positions & states
  update();
  // 3. Draw everything
  draw();
  // Repeat immediately ✅
  requestAnimationFrame(gameLoop);
}
gameLoop(); // Start the game!

Example — Moving Ball:

Halimbawa — gumagalaw at tumatalbog na bola 
let x = 50, y = 50;
let speedX = 3, speedY = 2;

function update() {
  x += speedX;
  y += speedY;
  // Bounce off edges
  if (x <= 0 || x >= canvas.width) speedX *= -1;
  if (y <= 0 || y >= canvas.height) speedY *= -1;
}

function draw() {
  ctx.beginPath();
  ctx.arc(x, y, 25, 0, Math.PI * 2);
  ctx.fillStyle = '#10b981';
  ctx.fill();
  ctx.closePath();
}

function gameLoop() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  update();
  draw();
  requestAnimationFrame(gameLoop);
}
gameLoop();

Run it → The ball moves and bounces forever! 🎉 Gagalaw at tatalbog sa gilid — patuloy hanggang isara ang pahina

📖 Chapter 3 — Player Input: Control the Game ⌨️

Kontrolin gamit ang keyboard

Capture keyboard keys so your character responds instantly!

// Track pressed keys
const keys = {};

// When key pressed
document.addEventListener('keydown', e => {
  keys[e.key] = true;
});
// When key released
document.addEventListener('keyup', e => {
  keys[e.key] = false;
});

// Player object
let player = { x: 100, y: 100, size: 40, speed: 5 };

function update() {
  if (keys['ArrowLeft']) player.x -= player.speed;
  if (keys['ArrowRight']) player.x += player.speed;
  if (keys['ArrowUp']) player.y -= player.speed;
  if (keys['ArrowDown']) player.y += player.speed;
  
  // Keep player inside screen
  player.x = Math.max(0, Math.min(canvas.width - player.size, player.x));
  player.y = Math.max(0, Math.min(canvas.height - player.size, player.y));
}

function draw() {
  ctx.fillStyle = '#10b981';
  ctx.fillRect(player.x, player.y, player.size, player.size);
}

Use arrow keys → Your character moves freely! ✅ Pindutin ang arrow keys — makagalaw nang hindi lumalabas sa screen

📖 Chapter 4 — Collision Detection: Detect Contact 💥

Alamin kung nagdikit ang mga bagay

How do you know when two objects touch? Here's the standard formula:

// Rectangle vs Rectangle collision
function rectCollision(a, b) {
  return (
    a.x < b.x + b.width &&
    a.x + a.width > b.x &&
    a.y < b.y + b.height &&
    a.y + a.height > b.y
  );
}

// Example — Collecting coins:
const player = { x: 50, y: 50, w: 40, h: 40 };
const coin = { x: 200, y: 150, w: 20, h: 20, collected: false };
let score = 0;

function update() {
  if (!coin.collected && rectCollision(
    {x:player.x, y:player.y, width:player.w, height:player.h},
    {x:coin.x, y:coin.y, width:coin.w, height:coin.h}
  )) {
    coin.collected = true;
    score += 10;
  }
}

function draw() {
  if (!coin.collected) {
    ctx.beginPath();
    ctx.arc(coin.x + 10, coin.y + 10, 10, 0, Math.PI * 2);
    ctx.fillStyle = '#f59e0b';
    ctx.fill();
  }
}

Touch the coin → Score increases & coin disappears! ✅ Kapag nagdikit — nadadagdagan ang puntos, nawawala ang barya

📖 Chapter 5 — Game States: Start · Play · Game Over 📊

Iba't ibang pahina — hindi lang naglalaro

Separate screens for the beginning, during play, and when the game ends.

let gameState = 'start'; // 'start' | 'playing' | 'gameover'
let score = 0;

function drawStartScreen() {
  ctx.fillStyle = '#fff';
  ctx.font = '40px Arial';
  ctx.textAlign = 'center';
  ctx.fillText('🎮 MY ADVENTURE GAME', canvas.width/2, canvas.height/2 - 40);
  ctx.font = '24px Arial';
  ctx.fillText('Press SPACE to Start!', canvas.width/2, canvas.height/2 + 20);
}

function drawGameScreen() {
  ctx.font = '20px Arial';
  ctx.fillText('Score: ' + score, 700, 30);
  // Draw player, enemies, coins here
}

function drawGameOverScreen() {
  ctx.fillStyle = '#ef4444';
  ctx.font = '40px Arial';
  ctx.fillText('GAME OVER 💀', canvas.width/2, canvas.height/2 - 40);
  ctx.fillStyle = '#fff';
  ctx.font = '24px Arial';
  ctx.fillText('Final Score: ' + score, canvas.width/2, canvas.height/2 + 10);
  ctx.fillText('Press R to Restart', canvas.width/2, canvas.height/2 + 50);
}

// Route drawing based on current state
function gameLoop() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  if (gameState === 'start') {
    drawStartScreen();
  } else if (gameState === 'playing') {
    update();
    drawGameScreen();
  } else if (gameState === 'gameover') {
    drawGameOverScreen();
  }
  requestAnimationFrame(gameLoop);
}

// Change state with keys
document.addEventListener('keydown', e => {
  if (e.key === ' ' && gameState === 'start') gameState = 'playing';
  if (e.key === 'r' && gameState === 'gameover') {
    gameState = 'playing';
    score = 0;
    // Reset player & enemies here
  }
});

let frame = 0;
let frameTimer = 0;

function update() {
  frameTimer++;
  if (frameTimer >= 10) { // Change frame every 10 loops
    frame = (frame + 1) % 4; // cycles: 0, 1, 2, 3 → back to 0
    frameTimer = 0;
  }
}
